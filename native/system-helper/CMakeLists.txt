cmake_minimum_required(VERSION 3.24)
cmake_policy(SET CMP0091 NEW)
project(bandbuddy_system_helper LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(bandbuddy-system-helper main.cpp)
target_compile_features(bandbuddy-system-helper PRIVATE cxx_std_17)
target_compile_options(bandbuddy-system-helper PRIVATE /utf-8 /W4)
target_link_libraries(bandbuddy-system-helper PRIVATE wintrust crypt32 shell32 advapi32)
install(TARGETS bandbuddy-system-helper RUNTIME DESTINATION .)
